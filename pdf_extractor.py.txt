import os

# 1. CONTROLLO DIPENDENZE (Diagnostica)
try:
    import fitz  # PyMuPDF
    has_fitz = True
except ImportError:
    has_fitz = False

try:
    import pypdf
    has_pypdf = True
except ImportError:
    has_pypdf = False

print(f"PyMuPDF (fitz) installed: {has_fitz}")
print(f"pypdf installed: {has_pypdf}")

# 2. CONTROLLO FILE NELLA CARTELLA
files = os.listdir('.')
print(f"Files in dir: {files}")


# 3. FUNZIONE DI ESTRAZIONE TESTO DA PDF
def estrai_testo_pdf(pdf_path):
    testo_estratto = ""
    
    if has_fitz:
        print(f"\n[INFO] Utilizzo PyMuPDF (fitz) per: {pdf_path}")
        doc = fitz.open(pdf_path)
        for num_pagina in range(len(doc)):
            pagina = doc[num_pagina]
            testo_estratto += f"\n--- Pagina {num_pagina + 1} ---\n"
            testo_estratto += pagina.get_text()
        doc.close()
        
    elif has_pypdf:
        print(f"\n[INFO] Utilizzo pypdf per: {pdf_path}")
        reader = pypdf.PdfReader(pdf_path)
        for i, pagina in enumerate(reader.pages):
            testo_estratto += f"\n--- Pagina {i + 1} ---\n"
            testo_estratto += pagina.extract_text() or ""
            
    else:
        print("\n[ERRORE] Nessuna libreria PDF installata. Installa PyMuPDF o pypdf!")
        return None
        
    return testo_estratto


# 4. ESECUZIONE SUL PRIMO PDF TROVATO
pdf_files = [f for f in files if f.lower().endswith('.pdf')]

if pdf_files:
    primo_pdf = pdf_files[0]
    testo = estrai_testo_pdf(primo_pdf)
    
    if testo:
        print("\n=== ANTEPRIMA TESTO ESTRATTO (primi 500 caratteri) ===")
        print(testo[:500])
else:
    print("\n[AVVISO] Nessun file .pdf trovato nella cartella corrente.")
