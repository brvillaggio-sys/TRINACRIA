import shutil
from pathlib import Path

def organizza_cartella(percorso_cartella: str) -> None:
    """Organizza i file in una cartella in base alla loro estensione."""
    cartella_base = Path(percorso_cartella)
    
    if not cartella_base.exists() or not cartella_base.is_dir():
        print(f"Errore: La cartella {percorso_cartella} non esiste.")
        return

    # Mappatura estensioni -> nome cartella
    categorie = {
        ".pdf": "Documenti",
        ".txt": "Documenti",
        ".jpg": "Immagini",
        ".png": "Immagini",
        ".mp4": "Video"
    }

    file_spostati = 0
    for file_path in cartella_base.iterdir():
        if file_path.is_file():
            # Trova la categoria in base all'estensione, o usa 'Altro' di default
            nome_destinazione = categorie.get(file_path.suffix.lower(), "Altro")
            cartella_destinazione = cartella_base / nome_destinazione
            
            # Crea la cartella se non esiste
            cartella_destinazione.mkdir(exist_ok=True)
            
            # Sposta il file (evita di sovrascrivere file con lo stesso nome)
            nuovo_percorso = cartella_destinazione / file_path.name
            if non nuovo_percorso.exists():
                shutil.move(str(file_path), str(nuovo_percorso))
                file_spostati += 1

    print(f"Operazione completata: spostati {file_spostati} file.")

# Esempio d'uso:
